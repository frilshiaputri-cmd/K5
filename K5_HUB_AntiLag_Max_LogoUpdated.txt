local function MakeDraggable(Object)

	local dragging = false
	local dragStart
	local startPosition

	Object.InputBegan:Connect(function(input)

		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			dragStart = input.Position
			startPosition = Object.Position

			input.Changed:Connect(function()

				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end

			end)

		end

	end)

	UserInputService.InputChanged:Connect(function(input)

		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		Object.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)

	end)

end

--========================================================--
-- ANTI AFK / AUTO ATTACK CLICK
-- Klik tombol Attack seperti tombol pedang pada layar
-- setiap 60 detik agar karakter tidak dianggap idle.
--========================================================--

local ANTI_AFK_INTERVAL = 60

local function ClickAttackButton()
    local Camera = workspace.CurrentCamera
    if not Camera then return false end

    -- Coba cari tombol Attack melalui nama GUI terlebih dahulu.
    local PlayerGui = Player:FindFirstChildOfClass("PlayerGui")
    if PlayerGui then
        local keywords = {
            "attack", "basicattack", "basic_attack",
            "normalattack", "normal_attack",
            "serang", "pedang", "sword"
        }

        for _, obj in ipairs(PlayerGui:GetDescendants()) do
            if obj:IsA("GuiButton") and obj.Visible then
                local name = string.lower(obj.Name or "")
                local match = false

                for _, keyword in ipairs(keywords) do
                    if string.find(name, keyword, 1, true) then
                        match = true
                        break
                    end
                end

                if match then
                    local fired = pcall(function()
                        for _, connection in ipairs(getconnections(obj.Activated)) do
                            connection:Fire()
                        end
                    end)

                    if fired then
                        return true
                    end
                end
            end
        end
    end

    -- Fallback untuk tombol Attack pada posisi kanan-bawah layar.
    local viewport = Camera.ViewportSize
    local x = viewport.X * 0.89
    local y = viewport.Y * 0.61

    local ok = pcall(function()
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
        task.wait(0.08)
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)

    return ok
end

local function StartAntiAFK()
    if AntiAFKRunning then return end
    AntiAFKRunning = true

    task.spawn(function()
        -- Klik pertama setelah 60 detik.
        while AntiAFK do
            task.wait(ANTI_AFK_INTERVAL)

            if not AntiAFK then
                break
            end

            local Character = Player.Character
            local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                ClickAttackButton()
            end
        end

        AntiAFKRunning = false
    end)
end

local AntiLagConnections = {}

local function DisableVisualEffect(obj)
    pcall(function()
        if obj:IsA("ParticleEmitter")
            or obj:IsA("Trail")
            or obj:IsA("Beam")
            or obj:IsA("Smoke")
            or obj:IsA("Fire")
            or obj:IsA("Sparkles") then
            obj.Enabled = false

        elseif obj:IsA("PostEffect") then
            obj.Enabled = false

        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            -- Hilangkan gambar/tekstur dekoratif agar beban render lebih ringan.
            obj.Transparency = 1

        elseif obj:IsA("BasePart") then
            -- Turunkan beban material, refleksi dan bayangan di client.
            obj.CastShadow = false
            obj.Reflectance = 0

            if obj.Material ~= Enum.Material.SmoothPlastic
                and obj.Material ~= Enum.Material.Plastic then
                obj.Material = Enum.Material.SmoothPlastic
            end
        end
    end)
end

local function ApplyAntiLag()
    if not AntiLag then return end

    local Lighting = game:GetService("Lighting")
    local Terrain = workspace:FindFirstChildOfClass("Terrain")

    -- Turunkan kualitas rendering/lighting.
    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.EnvironmentDiffuseScale = 0
        Lighting.EnvironmentSpecularScale = 0
        Lighting.Technology = Enum.Technology.Compatibility
    end)

    -- Matikan dekorasi Terrain bila tersedia.
    if Terrain then
        pcall(function()
            Terrain.Decoration = false
        end)

        pcall(function()
            Terrain.WaterWaveSize = 0
            Terrain.WaterWaveSpeed = 0
            Terrain.WaterReflectance = 0
            Terrain.WaterTransparency = 1
        end)
    end

    -- Turunkan Graphics Quality Roblox ke level rendah.
    pcall(function()
        local UserGameSettings = UserSettings():GetService("UserGameSettings")
        UserGameSettings.GraphicsQualityLevel = 1
    end)

    -- Bersihkan semua efek visual yang sudah ada.
    for _, obj in ipairs(workspace:GetDescendants()) do
        if not AntiLag then break end
        DisableVisualEffect(obj)
    end

    -- Matikan efek yang muncul setelah fitur dinyalakan.
    if AntiLagConnections.DescendantAdded then
        AntiLagConnections.DescendantAdded:Disconnect()
    end

    AntiLagConnections.DescendantAdded = workspace.DescendantAdded:Connect(function(obj)
        if AntiLag then
            task.defer(function()
                DisableVisualEffect(obj)
            end)
        end
    end)
end

local function StopAntiLag()
    if AntiLagConnections.DescendantAdded then
        AntiLagConnections.DescendantAdded:Disconnect()
        AntiLagConnections.DescendantAdded = nil
    end
end

--========================================================--
-- REDEEM
--========================================================--

local function RedeemCode(code)

	local success = pcall(function()

		CodeRE:FireServer({
			event = "usecode",
			code = code
		})

	end)

	if success then
		Redeemed += 1

		RedeemedValue.Text = tostring(Redeemed)

		local percent = Redeemed / #Codes

		TweenService:Create(
			Progress,
			TweenInfo.new(0.2),
			{
				Size = UDim2.new(percent, 0, 1, 0)
			}
		):Play()
	end

end

--========================================================--
-- AUTO REDEEM LOOP
--========================================================--

local function StartAutoRedeem()

	if RedeemRunning then
		return
	end

	RedeemRunning = true

	task.spawn(function()

		for index, code in ipairs(Codes) do

			if not AutoRedeem then
				break
			end

			Status.Text = "REDEEMING  •  " .. index .. "/" .. #Codes

			print(
				("[AUTO REDEEM] [%d/%d] %s")
					:format(index, #Codes, code)
			)

			RedeemCode(code)

			if index < #Codes and AutoRedeem then
				task.wait(REDEEM_DELAY)
			end

		end

		if AutoRedeem then
			Status.Text = "FINISHED"
		else
			Status.Text = "STOPPED"
		end

		RedeemRunning = false

	end)

end

	UpdateToggle()

	if AutoRedeem then
		Status.Text = "STARTING..."

	end

end)
