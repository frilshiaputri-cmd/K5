local function New(class, properties)
	local obj = Instance.new(class)

	for property, value in pairs(properties) do
		obj[property] = value
	end

	return obj
end

local function Corner(parent, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = parent
	return corner
end

local function Stroke(parent, color, thickness, transparency)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Thickness = thickness or 1
	stroke.Transparency = transparency or 0
	stroke.Parent = parent
	return stroke
end

local Running = false
local Redeemed = 0

local function SetStatus(index, status, color)
	if Rows[index] then
		Rows[index].Status.Text = "● " .. status
		Rows[index].Status.TextColor3 = color
	end
end

local function UpdateProgress(index)
	local percent = index / #Codes

	TweenService:Create(
		ProgressBar,
		TweenInfo.new(0.25, Enum.EasingStyle.Quad),
		{
			Size = UDim2.new(percent, 0, 1, 0)
		}
	):Play()

	ProgressValue.Text = index .. " / " .. #Codes
	RedeemedValue.Text = tostring(Redeemed)
end

local function RedeemCode(index)
	local code = Codes[index]

	SetStatus(index, "REDEEMING", Color3.fromRGB(255, 220, 50))

	local success = pcall(function()
		CodeRE:FireServer({
			event = "usecode",
			code = code
		})
	end)

	if success then
		Redeemed += 1
		SetStatus(index, "SENT", GREEN)
	else
		SetStatus(index, "ERROR", RED)
	end

	UpdateProgress(index)
end

-- Individual USE buttons
for index, rowData in pairs(Rows) do
	rowData.Use.MouseButton1Click:Connect(function()
		if Running then
			return
		end

		RedeemCode(index)
	end)
end

-- Redeem all
RedeemAll.MouseButton1Click:Connect(function()
	if Running then
		return
	end

	Running = true
	StatusValue.Text = "RUNNING"
	StatusValue.TextColor3 = Color3.fromRGB(255, 220, 50)

	RedeemAll.Text = "⏳   RUNNING..."

	task.spawn(function()
		for index = 1, #Codes do
			if not Running then
				break
			end

			print(("[AUTO REDEEM] [%d/%d] %s"):format(
				index,
				#Codes,
				Codes[index]
			))

			RedeemCode(index)

			if index < #Codes and Running then
				task.wait(REDEEM_DELAY)
			end
		end

		if Running then
			StatusValue.Text = "FINISHED"
			StatusValue.TextColor3 = GREEN
			RedeemAll.Text = "✓   FINISHED"
		else
			StatusValue.Text = "STOPPED"
			StatusValue.TextColor3 = RED
			RedeemAll.Text = "▶   REDEEM ALL"
		end

		Running = false
	end)
end)

-- Stop
Stop.MouseButton1Click:Connect(function()
	if Running then
		Running = false
		StatusValue.Text = "STOPPING..."
		StatusValue.TextColor3 = RED
		RedeemAll.Text = "▶   REDEEM ALL"
	end
end)
