--========================================================--
-- DRAG FUNCTION
--========================================================--

local function MakeDraggable(Object)

	local dragging = false
	local dragStart
	local startPosition

	Object.InputBegan:Connect(function(input)

		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			dragStart = input.Position
			startPosition = Object.Position

			input.Changed:Connect(function()

				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end

			end)

		end

	end)

	UserInputService.InputChanged:Connect(function(input)

		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		Object.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)

	end)

end

--========================================================--
-- UPDATE TOGGLE
--========================================================--

local function UpdateToggle()

	if AutoRedeem then

		Toggle.BackgroundColor3 = GREEN

		TweenService:Create(
			ToggleCircle,
			TweenInfo.new(0.15),
			{
				Position = UDim2.new(1, -22, 0, 3)
			}
		):Play()

	else

		Toggle.BackgroundColor3 = Color3.fromRGB(55, 60, 57)

		TweenService:Create(
			ToggleCircle,
			TweenInfo.new(0.15),
			{
				Position = UDim2.fromOffset(3, 3)
			}
		):Play()

	end

end

--========================================================--
-- REDEEM
--========================================================--

local function RedeemCode(code)

	local success = pcall(function()

		CodeRE:FireServer({
			event = "usecode",
			code = code
		})

	end)

	if success then
		Redeemed += 1

		RedeemedValue.Text = tostring(Redeemed)

		local percent = Redeemed / #Codes

		TweenService:Create(
			Progress,
			TweenInfo.new(0.2),
			{
				Size = UDim2.new(percent, 0, 1, 0)
			}
		):Play()
	end

end

--========================================================--
-- AUTO REDEEM LOOP
--========================================================--

local function StartAutoRedeem()

	if RedeemRunning then
		return
	end

	RedeemRunning = true

	task.spawn(function()

		for index, code in ipairs(Codes) do

			if not AutoRedeem then
				break
			end

			Status.Text = "REDEEMING  •  " .. index .. "/" .. #Codes

			print(
				("[AUTO REDEEM] [%d/%d] %s")
					:format(index, #Codes, code)
			)

			RedeemCode(code)

			if index < #Codes and AutoRedeem then
				task.wait(REDEEM_DELAY)
			end

		end

		if AutoRedeem then
			Status.Text = "FINISHED"
		else
			Status.Text = "STOPPED"
		end

		RedeemRunning = false

	end)

end

--========================================================--
-- TOGGLE
--========================================================--

Toggle.MouseButton1Click:Connect(function()

	AutoRedeem = not AutoRedeem

	UpdateToggle()

	if AutoRedeem then
		Status.Text = "STARTING..."
		StartAutoRedeem()
	else
		Status.Text = "AUTO REDEEM OFF"
	end

end)
